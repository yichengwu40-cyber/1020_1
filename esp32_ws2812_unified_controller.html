ESP32 Arduino 韌體程式碼 

#include <Adafruit_NeoPixel.h>

const int LDR_PIN = 39;     
const int POT_PIN = 34;     
const int LED_PIN = 26;     
const int NUM_PIXELS = 3;   

Adafruit_NeoPixel pixels(NUM_PIXELS, LED_PIN, NEO_GRB + NEO_KHZ800);

bool ledState = true;       
int mainRed = 255;          
int mainGreen = 0;          
int mainBlue = 127;         
int webBrightness = 255;    

void setup() {
  Serial.begin(115200);
  pixels.begin();
  pixels.show();
}

void loop() {
  if (Serial.available() > 0) {
    String command = Serial.readStringUntil('\n');
    command.trim();

    if (command == "ON") {
      ledState = true;
      Serial.println("ACK:ON");
    } 
    else if (command == "OFF") {
      ledState = false;
      Serial.println("ACK:OFF");
    }
    else if (command.startsWith("RGB:")) {
      int firstComma = command.indexOf(',');
      int secondComma = command.indexOf(',', firstComma + 1);
      if (firstComma > 0 && secondComma > 0) {
        mainRed = command.substring(4, firstComma).toInt();
        mainGreen = command.substring(firstComma + 1, secondComma).toInt();
        mainBlue = command.substring(secondComma + 1).toInt();
        ledState = true;
        Serial.printf("ACK:RGB(%d,%d,%d)\n", mainRed, mainGreen, mainBlue);
      }
    }
    else if (command.startsWith("BRIGHT:")) {
      webBrightness = command.substring(7).toInt();
      webBrightness = constrain(webBrightness, 0, 255);
      Serial.printf("ACK:BRIGHT(%d)\n", webBrightness);
    }
  }

  int ldrValue = analogRead(LDR_PIN);  
  int potValue = analogRead(POT_PIN);  

  float darkMultiplier = (float)(4095 - ldrValue) / 4095.0;
  if (darkMultiplier < 0.1) darkMultiplier = 0.1; 

  float potMultiplier = (float)potValue / 4095.0;
  float finalRatio = ((float)webBrightness / 255.0) * darkMultiplier * potMultiplier;

  for (int i = 0; i < NUM_PIXELS; i++) {
    if (!ledState) {
      pixels.setPixelColor(i, pixels.Color(0, 0, 0));
    } else {
      int r = mainRed * finalRatio;
      int g = mainGreen * finalRatio;
      int b = mainBlue * finalRatio;
      pixels.setPixelColor(i, pixels.Color(r, g, b));
    }
  }
  pixels.show();

  Serial.printf("{\"ldr\":%d,\"pot\":%d,\"state\":%s,\"bright\":%d}\n", 
                ldrValue, potValue, ledState ? "true" : "false", (int)(finalRatio * 255));

  delay(100);
}
